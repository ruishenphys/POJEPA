### 3.1 新模块（`src/po_jepa/runtime/addon/` 下）

#### 3.1.1 `calibrate.py`（174 行；16 §3.2）

- 方法和 v7 的 `workflow.calibrate_weights`（340 行）逐步相同：种子 1703；查询序列、学习率调度、裁剪和正式训练相同；先在初始状态下量前 `batches` 批的三项梯度范数（只记录），再用前 `pretrain_updates` 批只做 JEPA 训练，然后在紧接着的 `batches` 批上量三项梯度范数、取中位数，λ_c = ρ·median|∇L_JEPA| / median|∇L_c|。和 v7 不同的只有三处：不标定 σ_z（用导入的，`load_protocol()` 读，16 §3.2 第 1 步）；λ 的分母为 0 时报错（v7 没有这个判断，16 §3.2 第 5 步）；初始状态那一组 λ 的分母为 0 时记 `None`（transformer_consistency 在登记的初始化下一致性梯度恰为 0，S10b §9 第 13 条；v7 在这里会除以 0）。
- 只用 JEPA 的版本（16 §3.2 第 2 步）：transformer_consistency 直接用 `build_condition("matched_transformer")`（模型、初始权重和两个权重为 0 的损失配置都和 matched_transformer 相同）；fixed_fourier 用 `build_fixed_fourier(config["model"], model_seed(), device)`（S10b 建 fixed_fourier 条件用的同一个函数、同一个种子），损失配置是协议的 `LossConfig` 把两个权重置 0。不能直接用 `build_condition("fixed_fourier")`：它要先读补跑标定文件，而这个文件正是本命令要写的。

- 写文件的顺序：先查补跑标定文件不存在（存在就报错、不覆盖：它可能已经被训练任务读过，§9 第 7 条）；核导入的 v7 标定（`require_imported_v7_calibration`）；两个条件都算完以后再核一次 v7 标定没变；然后写两份 `calibration-<条件>.json`（`cache._write_json`）和补跑标定文件（同目录临时文件、fsync、硬链接发布，不覆盖）；写完马上用 S10b 的 `load_addon_weights` 读回（训练任务开始前做的就是这个核对），读回不对就删掉文件再报错。所以中途任何一步失败，补跑标定文件都不存在（14 §3 S17“两个条件都完成才写文件”）。

```python
def require_imported_v7_calibration() -> str:
    """补跑入口用的 v7 标定必须是 tools/import_v7_calibration.py 导入的那份（16 §3.1）：IMPORTED.json 里有这个文件名的
    记录，记录的 sha256 等于文件当前的 SHA。返回这个 SHA。"""
    path = check.v7_calibration_path()
    record = path.parent/IMPORT_RECORD
    if not path.exists():
        raise ValueError(f"缺少 v7 标定 {path}：先运行 python tools/import_v7_calibration.py")
    if not record.exists():
        raise ValueError(f"缺少 {record}：v7 标定必须用 tools/import_v7_calibration.py 导入，不能在本目录里重新标定")
    entry = read(record).get("files", {}).get(path.name)
    digest = cache.file_sha(path)
    if entry is None or entry.get("sha256") != digest:
        raise ValueError(f"{path.name} 和 {IMPORT_RECORD} 的导入记录不符：本目录的 v7 标定被改过或不是导入的")
    return digest


def _jepa_only(condition, device):
    """16 §3.2 第 2 步，S10b §4.1：返回 (模型, 两个权重为 0 的损失配置)。"""
    if condition == "fixed_fourier":
        config = check.load_protocol()
        loss = LossConfig(**{f.name: config["loss"][f.name] for f in fields(LossConfig)})
        return build_fixed_fourier(config["model"], check.model_seed(), device), replace(loss, time_weight=0., space_weight=0.)
    if condition == "transformer_consistency":
        return comparison.build_condition("matched_transformer", device)
    raise ValueError("only fixed_fourier and transformer_consistency use the add-on calibration")


def _gradient_norms(condition, model, batch, loss) -> dict:
    """三项梯度范数，键 jepa、time、space、grad_jepa、grad_time、grad_space（16 §3.2 第 5 步）。"""
    if condition == "transformer_consistency":
        return calibration.generic_gradient_norms(model, batch, loss_config=loss)
    return training.consistency_gradient_norms(model, batch, loss_config=loss)


def _pretrain_step(condition, model, optimizer, batch, loss, completed):
    """只做 JEPA 的一步：Transformer 走 matched_transformer 的单步，fixed_fourier 和 v7 的 calibrate 一样走 no_consistency。"""
    if condition == "transformer_consistency":
        return comparison.condition_step("matched_transformer", model, optimizer, batch, loss, completed)
    return training.train_step(model, optimizer, batch, loss_config=loss, completed_steps=completed, condition="no_consistency")


def _summary(rows) -> dict:
    return dict(median={k: statistics.median(r[k] for r in rows) for k in rows[0]},
                mean={k: sum(r[k] for r in rows)/len(rows) for k in rows[0]})


def calibrate_condition(condition, features, device) -> dict:
    """一个条件的标定记录。键是 v7 calibration.json 的键加 condition；v7_calibration_file、v4_overlay_sha256_after 等
    由 calibrate_addon 在两个条件都完成后补上。"""
    config = check.load_protocol()
    cal = config["calibration"]
    pre, count, rho = cal["pretrain_updates"], cal["batches"], cal["rho"]
    sequence, _ = comparison.learning.make_query_sequence(config)
    if pre+count > len(sequence):
        raise ValueError("pretrain_updates+batches exceeds the training sequence")
    model, loss = _jepa_only(condition, device)
    model.train()

    def rows(queries):
        return [_gradient_norms(condition, model, data.load_training_batch(features, q, device=str(device)), loss) for q in queries]

    initial = rows(sequence[:count])
    optimizer = comparison.make_optimizer(model)
    history = []
    for step in range(pre):
        batch = data.load_training_batch(features, sequence[step], device=str(device))
        result = _pretrain_step(condition, model, optimizer, batch, loss, step)
        history.append(dict(step=result.completed_steps, grad_norm_before_clip=result.grad_norm, **result.loss_values))
        if result.completed_steps % 50 == 0:
            print(f'calibrate-addon {condition} pretrain {result.completed_steps}/{pre} jepa={result.loss_values["jepa"]:.4f}', flush=True)
    measured = rows(sequence[pre:pre+count])
    after, before = _summary(measured), _summary(initial)
    median = after["median"]
    weights, initial_weights = {}, {}
    for name, key in (("time_weight", "grad_time"), ("space_weight", "grad_space")):
        if not median[key] > 0:
            raise ValueError(f"{condition}: median {key} = {median[key]!r}; λ 的分母为 0，不能标定（不除出 inf）")
        weights[name] = rho*median["grad_jepa"]/median[key]
        if not (math.isfinite(weights[name]) and weights[name] > 0):
            raise ValueError(f"calibrated weight must be finite and positive: {condition}.{name}")
        # 初始状态只记录：transformer_consistency 的 p_out 零初始化，一致性梯度恰为 0，记 None（S10b §9 第 13 条）
        initial_weights[name] = rho*before["median"]["grad_jepa"]/before["median"][key] if before["median"][key] > 0 else None
    graph = config["model"]["graph"]
    return dict(condition=condition, rho=rho, batches=len(measured), statistic="median", method="v6_after_jepa_pretraining",
                pretrain_updates=pre, pretrain_history=history, per_batch=measured, median=median, mean=after["mean"],
                initial_state=dict(per_batch=initial, **before, **initial_weights), **weights,
                graph=dict(sigma_x=graph["sigma_x"], sigma_z=graph["sigma_z"], max_modes=graph["max_modes"]),
                v4_overlay_sha256_before=cache.file_sha(OVERLAY),
                note="仿真补跑标定；λ 写进 configs/calibration/addon/，三个种子共用")


def _relative(path: Path) -> str:
    return path.relative_to(ROOT).as_posix() if path.is_relative_to(ROOT) else str(path)


def _publish(path: Path, text: str) -> None:
    """同目录临时文件写完、fsync，再硬链接发布（目标已存在就失败，不覆盖）。"""
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(f".{path.name}.{os.getpid()}.tmp")
    try:
        with temporary.open("x", encoding="utf-8") as handle:
            handle.write(text)
            handle.flush()
            os.fsync(handle.fileno())
        os.link(temporary, path)
    finally:
        temporary.unlink(missing_ok=True)


def calibrate_addon(features, output, device) -> dict:
    """16 §3.2：output 是已建好的空目录。依次标定 CALIBRATED_CONDITIONS，全部完成后写记录和补跑标定文件，返回写入的内容。"""
    path = calibration.addon_calibration_path()
    if path.exists():
        raise ValueError(f"补跑标定文件已存在：{path}。它可能已被训练任务用过，不覆盖；确实要重标定请先告诉 Jerry")
    v7_file = check.v7_calibration_path()
    v7_sha = require_imported_v7_calibration()
    results = {c: calibrate_condition(c, Path(features), device) for c in calibration.CALIBRATED_CONDITIONS}
    if require_imported_v7_calibration() != v7_sha:
        raise ValueError("v7 calibration file changed during calibrate-addon")
    protocol = check.load_protocol()
    source = (f"calibrate-addon：JEPA 预训练 {protocol['calibration']['pretrain_updates']} 步后，"
              f"{protocol['calibration']['batches']} 批梯度范数的中位数；{output.parent.name}")
    payload = {c: dict(rho=r["rho"], time_weight=r["time_weight"], space_weight=r["space_weight"], source=source)
               for c, r in results.items()}
    payload.update(sigma_z_used=protocol["model"]["graph"]["sigma_z"], v7_calibration_sha256=v7_sha)
    header = f"# 仿真补跑标定（run.py calibrate-addon 自动写入，勿手改）：{data.datasets.active().tag}，{check.v7_updates()} 步\n"
    text = header+yaml.safe_dump(payload, allow_unicode=True, sort_keys=True)
    if yaml.safe_load(text) != payload:
        raise ValueError("add-on calibration YAML does not round-trip")
    for c, r in results.items():
        cache._write_json(output/f"calibration-{c}.json", dict(r, v7_calibration_file=_relative(v7_file),
            v7_calibration_sha256=v7_sha, addon_calibration_file=_relative(path), v4_overlay_sha256_after=cache.file_sha(OVERLAY)))
    _publish(path, text)
    try:
        for c in calibration.CALIBRATED_CONDITIONS:      # 训练任务读这个文件时的核对（S10b §4.2）必须通过
            if calibration.load_addon_weights(c) != dict(time_weight=results[c]["time_weight"], space_weight=results[c]["space_weight"]):
                raise ValueError("add-on calibration file reads back different weights")
    except Exception:
        path.unlink()
        raise
    print(yaml.safe_dump(payload, allow_unicode=True, sort_keys=True), flush=True)
    print(f"已写入 {_relative(path)}，可以开始 D 段。", flush=True)
    return payload
```

#### 3.1.2 `check_addon.py`（114 行；16 §5）

- 对 `ADDON_CONDITIONS` 的每个条件：`build_condition` 构造，记初始哈希（`v7jobs._model_hash`）；`comparison._validate_registration` 核参数量和预登记（koopman 的 K 等于按规则算的值、参数量等于公式，S10b）；在 v7 check 的同一批 4 个窗口（轨迹 0–3、起点 5、时长 1、2、4、8）上做一次 `comparison._observed_update`，读出次数等于 `addon.conditions.expected_readouts`（time_only、space_only、fixed_fourier 10 次，transformer_consistency 6 次，koopman 2 次），之后每个参数的梯度都存在且有限，记梯度和更新量的统计；再新建一个模型，用 v7 check 的查询序列前 100 项做 100 次正常更新并计时（`TIMED_UPDATES`）。
- 初始权重关系（08 §1.3）：time_only、space_only、fixed_fourier 的初始哈希等于 po_jepa_full 的，transformer_consistency 的等于 matched_transformer 的，koopman 只记录；不等就报错。
- 依赖版本和设备由入口先核（§3.1.3）。报告字段见 §4.3。

```python
"""16 §5：run.py check-addon——5 个新条件在真实特征上各做一次检查，写 <输出>/check-addon.json。

每个条件：comparison.build_condition 构造；初始权重哈希（v7jobs._model_hash）和 08 §1.3 的关系（time_only、space_only、
fixed_fourier 等于 po_jepa_full，transformer_consistency 等于 matched_transformer，koopman 只记录）；参数量和预登记
（comparison._validate_registration，koopman 另记 K）；v7 check 的同一批 4 个窗口上的一次真实更新
（comparison._observed_update：读出次数等于 addon.conditions.expected_readouts），之后每个参数的梯度都存在且有限；
再用新建的模型做 TIMED_UPDATES 次正常更新（v7 check 的同一个 100×4 查询序列）并计时。依赖和设备由 entry 先核。
"""

import math

import platform

import sys

import time

import numpy as np

import torch

from .. import cache, checks as check, comparison, data, resources, v7jobs
from . import conditions

TIMED_UPDATES = 100
REFERENCE = dict(po_jepa_time_only="po_jepa_full", po_jepa_space_only="po_jepa_full", fixed_fourier="po_jepa_full",
                 transformer_consistency="matched_transformer", koopman=None)


def initial_relations(hashes: dict) -> dict:
    """08 §1.3：hashes 含 7 个名字（5 个新条件和两个参照）的初始哈希。返回每个新条件的参照名（koopman 为 None）；
    不相等就报错。"""
    for condition, reference in REFERENCE.items():
        if reference is not None and hashes[condition] != hashes[reference]:
            raise ValueError(f"{condition} 的初始权重和 {reference} 不同（08 §1.3）")
    return dict(REFERENCE)


def _diagnostic(condition, features, device, queries, registration, sequence):
    """一次真实更新：读出次数、每个参数的梯度和更新量、参数量和预登记。返回 (初始哈希, 记录)。"""
    model, loss = comparison.build_condition(condition, device)
    initial = v7jobs._model_hash(model)
    comparison._validate_registration(condition, model, sequence, registration)
    optimizer = comparison.make_optimizer(model)
    batch = data.load_training_batch(features, queries, device=str(device))
    before = {name: p.detach().cpu().clone() for name, p in model.named_parameters()}
    if device.type == "cuda":
        torch.cuda.reset_peak_memory_stats(device)
    result, scales = comparison._observed_update(condition, model, optimizer, batch, loss, 0)
    resources.synchronize(device)
    memory = check._memory(device) if device.type == "cuda" else None
    parameters = {}
    for name, p in model.named_parameters():
        if p.grad is None or not torch.isfinite(p.grad).all():
            raise ValueError(f"{condition}: parameter {name} has no finite gradient")
        parameters[name] = dict(gradient=check.tensor_stats(p.grad), update=check.tensor_stats(p.detach().cpu()-before[name]))
    koopman = condition == "koopman"
    record = dict(loss_label=conditions.LOSS_LABELS[condition], time_weight=loss.time_weight, space_weight=loss.space_weight,
                  parameters=comparison.parameter_counts(model), koopman_modes=model.config.modes if koopman else None,
                  initial_model_sha256=initial, readouts=dict(observed=len(scales), expected=conditions.expected_readouts(condition, model)),
                  losses=result.loss_values, grad_max_abs=result.grad_max_abs, grad_norm=result.grad_norm,
                  gradients=parameters, memory=memory)
    return initial, record


def _timed(condition, features, device):
    """TIMED_UPDATES 次正常更新（不挂钩子、不做诊断），v7 check 的 100×4 查询序列；返回耗时和每步损失。"""
    model, loss = comparison.build_condition(condition, device)
    optimizer = comparison.make_optimizer(model)
    sequence = check.make_query_sequence()[:TIMED_UPDATES]
    if device.type == "cuda":
        torch.cuda.reset_peak_memory_stats(device)
    resources.synchronize(device)
    started, losses = time.perf_counter(), []
    for completed, rows in enumerate(sequence):
        batch = data.load_training_batch(features, rows, device=str(device))
        result = comparison.condition_step(condition, model, optimizer, batch, loss, completed)
        if result.completed_steps != completed+1 or any(
                value is not None and not math.isfinite(value) for value in result.loss_values.values()):
            raise ValueError(f"{condition}: timed update count or finite losses failed")
        losses.append(result.loss_values["jepa"])
        del batch
    resources.synchronize(device)
    seconds = time.perf_counter()-started
    if not (math.isfinite(seconds) and seconds > 0):
        raise ValueError("nonfinite update timing")
    return dict(updates=len(sequence), seconds=seconds, seconds_per_update=seconds/len(sequence), jepa=losses,
                memory=check._memory(device) if device.type == "cuda" else None)


def check_addon(features, output, device) -> dict:
    """output 是已建好的目录；写 output/check-addon.json 并返回其内容。"""
    with resources.isolated_randomness(device):
        hardware = check.device_metadata(device, output) if device.type == "cuda" else dict(device="cpu")
        assets = data.approved_cache_metadata(features)
        registration, sequence, _ = comparison.preregister()
        queries = [data.WindowQuery(i, 5, d) for i, d in enumerate((1., 2., 4., 8.))]   # 和 v7 check 的诊断批相同
        hashes = {name: v7jobs._model_hash(comparison.build_condition(name, device)[0])
                  for name in ("po_jepa_full", "matched_transformer")}
        records = {}
        for condition in conditions.ADDON_CONDITIONS:
            print(f"check-addon {condition}", flush=True)
            hashes[condition], records[condition] = _diagnostic(condition, features, device, queries, registration, sequence)
            records[condition]["timed"] = _timed(condition, features, device)
        relations = initial_relations(hashes)
        for condition, reference in relations.items():
            records[condition]["initial_reference"] = reference
    report = dict(hardware=hardware, cache_identity=assets, conditions=records,
                  reference_initial_model_sha256={name: hashes[name] for name in ("po_jepa_full", "matched_transformer")},
                  parameter_match=registration["parameter_match"], timed_updates=TIMED_UPDATES,
                  python=sys.version, torch=torch.__version__, cuda=torch.version.cuda, numpy=np.__version__,
                  platform=platform.platform(), cpu_threads=torch.get_num_threads(), check_completed=True)
    cache._write_json(output/"check-addon.json", report)
    return report
```

```python
"""run.py 的补跑子命令 calibrate-addon、check-addon（16 §3.2、§5）。v7 的 cli.py 和 workflow.execute 不动，
run.py 把这两个子命令交给本模块（python -m po_jepa.runtime.addon.entry …），参数和 cli.py 的相同。

--output 是 run.py 传来的 <输出>/run，必须不存在：
  calibrate-addon：calibration-fixed_fourier.json、calibration-transformer_consistency.json、COMPLETE.json，
                   另写 configs/calibration/addon/<tag>-u<步数>.yaml（和 v7 的 calibrate 一样不核 CUDA 设备）
  check-addon：environment.json、check-addon.json、gpu-inventory-*、COMPLETE.json（要求 cuda:N，和 v7 的 check 相同）
失败时写 FAILED.json（字段和 workflow.execute 的相同）并以非零退出码结束。
"""
import argparse
import os
from pathlib import Path
import traceback

COMMANDS = ("calibrate-addon", "check-addon")


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("command", choices=COMMANDS)
    parser.add_argument("--features-root", type=Path, required=True)
    parser.add_argument("--device", default="cuda:0")
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--seed", type=int, choices=(1703,), default=1703, help="只用种子 1703（三个种子共用一组 λ，和 v7 的 calibrate 相同）")
    args = parser.parse_args(argv)
    from ..checks import SEED_ENV
    os.environ[SEED_ENV] = str(args.seed)
    return execute(args)


def execute(args):
    import torch
    from .. import cache, checks, environment, workflow
    from . import calibrate, calibration
    if not checks.is_v7():
        raise ValueError("补跑子命令只用于 v7 协议（run.py 会设 PO_JEPA_PROTOCOL=v7）")
    bundle = workflow.verify_bundle()
    device = torch.device(args.device)
    if args.command == "check-addon" and (device.type != "cuda" or device.index is None):
        raise ValueError("explicit cuda:N target required")
    output = Path(args.output).resolve()
    output.mkdir(parents=True, exist_ok=False)
    stage = "environment"
    try:
        if args.command == "check-addon":
            cache._write_json(output/"environment.json", environment.verify_versions())
        stage = "calibration"
        workflow.require_calibrated_weights(swept=False)
        calibrate.require_imported_v7_calibration()
        if args.command == "calibrate-addon":
            stage = "calibrate-addon"
            calibrate.calibrate_addon(Path(args.features_root), output, device)
            path = calibration.addon_calibration_path()
            extra = dict(addon_calibration_file=calibrate._relative(path), addon_calibration_sha256=cache.file_sha(path))
        else:
            calibration.require_addon_calibration()
            stage = "check-addon"
            from .check_addon import check_addon
            report = check_addon(Path(args.features_root), output, device)
            extra = dict(check_sha256=cache.file_sha(output/"check-addon.json"), real_device=report["hardware"])
        if workflow.verify_bundle() != bundle:
            raise ValueError("source bundle changed during run")
        cache._write_json(output/"COMPLETE.json", dict(command=args.command, manifest_sha256=bundle, **extra,
                                                       scientific_conclusions_verified=False))
        return 0
    except Exception as error:
        cache._write_json(output/"FAILED.json", dict(stage=stage, error_type=type(error).__name__, message=str(error),
                                                     traceback=traceback.format_exc(), automatic_retry=False))
        raise


if __name__ == "__main__":
    raise SystemExit(main())
```

```python

def v7_head(tree: Path) -> str:
    result = subprocess.run(["git", "-C", str(tree), "rev-parse", "HEAD"], stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                            text=True, encoding="utf-8")
    if result.returncode:
        raise ValueError(f"读不到 {tree} 的 git HEAD：{result.stderr.strip()}")
    return result.stdout.strip()


def _check_values(path: Path, raw: bytes) -> None:
    values = yaml.safe_load(raw.decode("utf-8"))
    if type(values) is not dict:
        raise ValueError(f"{path} 不是 YAML 映射")
    for key in REQUIRED:
        value = values.get(key)
        if type(value) not in (int, float) or not math.isfinite(value) or value <= 0:
            raise ValueError(f"{path} 的 {key} 是 {value!r}，应为有限正数（v7 的 calibrate 没有完成？）")


def _replace_json(path: Path, value: dict) -> None:
    """cache._write_json 写到同目录的新临时名，再整体换名（目标可以已存在）。"""
    temporary = path.with_name(f".{path.name}.{os.getpid()}.tmp")
    for stale in (temporary, temporary.with_name(temporary.name+".partial")):
        stale.unlink(missing_ok=True)
    cache._write_json(temporary, value)
    os.replace(temporary, path)


def _copy(source_bytes: bytes, target: Path) -> None:
    """同目录临时文件写完、fsync，再硬链接到目标（目标已存在就失败，不覆盖）。"""
    temporary = target.with_name(f".{target.name}.{os.getpid()}.tmp")
    try:
        with temporary.open("xb") as handle:
            handle.write(source_bytes)
            handle.flush()
            os.fsync(handle.fileno())
        os.link(temporary, target)
    finally:
        temporary.unlink(missing_ok=True)


def import_calibration(v7_tree, stage, target, *, now=None) -> dict:
    """把本阶段的 v7 标定导入 target（正式用 ROOT/configs/calibration）；返回 {文件名: "imported" | "unchanged"}。"""
    v7_tree, target = Path(v7_tree).resolve(), Path(target)
    head = v7_head(v7_tree)
    if head != V7_COMMIT:
        raise ValueError(f"v7 目录 {v7_tree} 的 git HEAD 是 {head}，应为 {V7_COMMIT}")
    record_path = target/"IMPORTED.json"
    record = read(record_path) if record_path.exists() else dict(schema=SCHEMA, files={})
    if record.get("schema") != SCHEMA or type(record.get("files")) is not dict:
        raise ValueError(f"{record_path} 不是导入记录（schema {SCHEMA}）")
    plan = {}
    for file in file_names(stage):
        source = v7_tree/"configs/calibration"/file
        if not source.is_file():
            raise ValueError(f"缺少 v7 标定 {source}（v7 的 C 段没有跑完？）")
        raw = source.read_bytes()
        _check_values(source, raw)
        digest = hashlib.sha256(raw).hexdigest()
        destination = target/file
        if destination.exists() and destination.read_bytes() != raw:
            raise ValueError(f"{destination} 已存在且和 {source} 不同；不覆盖")
        entry = record["files"].get(file)
        if entry is not None and (entry.get("sha256") != digest or entry.get("v7_commit") != V7_COMMIT):
            raise ValueError(f"{record_path} 里 {file} 的记录和 {source} 不符")
        plan[file] = (source, raw, digest, destination, entry)
    stamp = now or datetime.now(timezone.utc).isoformat(timespec="seconds")
    status, added = {}, {}
    target.mkdir(parents=True, exist_ok=True)
    for file, (source, raw, digest, destination, entry) in plan.items():
        status[file] = "unchanged" if destination.exists() else "imported"
        if status[file] == "imported":
            _copy(raw, destination)
        if entry is None:
            added[file] = dict(source=str(source), sha256=digest, imported_at=stamp, v7_commit=V7_COMMIT)
    if added:
        _replace_json(record_path, dict(record, files={**record["files"], **added}))
    return status


def main(argv=None):
    p = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    p.add_argument("--v7-tree", type=Path, required=True, help="v7 目录（工作目录/po-jepa）")
    p.add_argument("--stage", choices=tuple(STAGES), required=True)
    a = p.parse_args(argv)
    status = import_calibration(a.v7_tree, a.stage, ROOT/"configs/calibration")
    for file, state in status.items():
        print(f"{file}：{'已导入' if state == 'imported' else '已存在且字节相同，跳过'}", flush=True)
    print(f"完成：{len(status)} 个标定文件（configs/calibration/，记录在 IMPORTED.json）", flush=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
    """--v6-features 的目录：每条轨迹的三个文件都在，首尾两条用 v6 的逐条读法（cache._read_complete_cache_uncached：
    核 manifest、来源、raw_z 的 SHA、shape 和有限性）完整读一遍。git 克隆里的 features/ 只有提交进仓库的元数据
    （report.json、COMPLETE.json、registration.json、0000/manifest.json），编码报告和缓存身份都对得上，只有这里拦得下。"""
    missing = [f"{tid:04d}" for tid in range(n) if not all((root/f"{tid:04d}"/name).is_file() for name in V6_FILES)]
    if missing:
        problems.append(f"特征 {root}：{len(missing)} 条轨迹缺 COMPLETE.json、manifest.json 或 raw_z.npy（第一条是 {missing[0]}）；"
                        "这不是完整的 v6 特征目录（git 克隆里的 features/ 只有元数据）")
        return False
    for tid in (0, n-1):
        try:
            cache._read_complete_cache_uncached(root/f"{tid:04d}")
        except Exception as error:
            problems.append(f"特征 {root}：轨迹 {tid:04d} 用 v6 的读法读不出来：{type(error).__name__}: {error}")
            return False
    return True


def scene_identity(features: Path, scene: str, n: int, problems: list, *, v6=False):
    """场景的特征缓存身份（v7 任务 preflight 记的同一个函数），并逐条核 manifest.json；v6=True 时另核 v6_trajectories。
    出错记进 problems，返回 None。"""
    system, knobs = split_scene(scene)
    old = datasets._set_for_tests(datasets.DatasetSpec(system, n, knobs))
    try:
        identity = data.approved_cache_metadata(features)
        if v6 and not v6_trajectories(features, n, problems):
            return None
    except Exception as error:
        problems.append(f"特征 {features}：{type(error).__name__}: {error}")
        return None
    finally:
        datasets._set_for_tests(old)
    for tid, digest in identity["manifest_sha256_by_id"].items():
        path = features/f"{tid:04d}"/"manifest.json"
        if not path.is_file() or cache.file_sha(path) != digest:
            problems.append(f"特征 {features}：轨迹 {tid:04d} 的 manifest.json 缺失或和编码报告不符")
    return _normal(identity)


def preflight(v7_results, features, stage, v6_features=None) -> dict:
    """返回要写的记录；有问题就把全部问题列出来后报 ValueError。"""
    spec = STAGES[stage]
    v7_results, features = Path(v7_results).resolve(), Path(features).resolve()
    v6 = Path(v6_features).resolve() if v6_features else None
    problems, scenes = [], {}
    for scene in spec["scenes"]:
        from_v6 = uses_v6(scene, spec["n"], v6)
        root = v6 if from_v6 else features/f"{name(scene)}-{spec['n']}"
        identity = scene_identity(root, scene, spec["n"], problems, v6=from_v6)
        tag = f"{name(scene)}-{spec['n']}"
        jobs = {}
        for seed in spec["seeds"]:
            for condition in V7_CONDITIONS:
                path = v7_results/name(scene)/f"seed{seed}"/condition
                done = finished_attempt(path)
                if done is None:
                    problems.append(f"v7 任务没有完成：{path}（含 -resumeN）")
                    continue
                job_dir = v7jobs._job_dir(done)
                job = read(job_dir/"job.json")
                if (job.get("condition"), job.get("dataset"), job.get("seed"), job.get("updates")) != (
                        condition, tag, seed, spec["updates"]):
                    problems.append(f"v7 任务记录对不上：{job_dir/'job.json'}")
                    continue
                recorded = read(job_dir/"report.json")["preflight"]["cache_identity"]
                if identity is not None and recorded != identity:
                    problems.append(f"v7 任务 {job_dir} 记的特征缓存身份和 {root} 现在的不同")
                jobs[f"seed{seed}/{condition}"] = os.path.relpath(job_dir, v7_results)
        if identity is not None:
            scenes[name(scene)] = dict(features=str(root), encode_report_sha256=identity["encode_report_sha256"],
                                       cache_identity_sha256=hashlib.sha256(cache._json_bytes(identity)).hexdigest(),
                                       trajectories=len(identity["manifest_sha256_by_id"]), v7_jobs=jobs)
    if problems:
        for line in problems:
            print("  "+line, flush=True)
        raise ValueError(f"preflight 没通过：{len(problems)} 项（见上），补跑停在 A0")
    return dict(schema=SCHEMA, stage=stage, v7_results=str(v7_results), features=str(features), v6_features=str(v6) if v6 else None,
                v7_conditions=list(V7_CONDITIONS), seeds=list(spec["seeds"]), scenes=scenes,
                checked_jobs=sum(len(s["v7_jobs"]) for s in scenes.values()))


def write_once(path: Path, value: dict) -> str:
    """不存在就用 cache._write_json 写；已存在且字节相同就跳过；不同就报错。"""
    path = Path(path)
    if path.exists():
        if path.read_bytes() != cache._json_bytes(value):
            raise ValueError(f"{path} 已存在且内容不同；不覆盖")
        return "unchanged"
    path.parent.mkdir(parents=True, exist_ok=True)
    path.with_name(path.name+".partial").unlink(missing_ok=True)
    cache._write_json(path, value)
    return "written"


def main(argv=None):
    p = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    p.add_argument("--v7-results", type=Path, required=True, help="results-v7/<pilot|full>")
    p.add_argument("--features", type=Path, required=True, help="features-v7")
    p.add_argument("--v6-features", type=Path, default=None,
                   help="试跑：v7 试跑用的 v6 double_well-200 特征目录（绝对路径，和 make_v7_jobs.py --v6-features 给的相同）")
    p.add_argument("--stage", choices=tuple(STAGES), required=True)
    p.add_argument("--output", type=Path, required=True)
    a = p.parse_args(argv)
    record = preflight(a.v7_results, a.features, a.stage, a.v6_features)
    state = write_once(a.output, record)
    print(f"preflight 通过：{len(record['scenes'])} 个场景，{record['checked_jobs']} 个 v7 任务；"
          f"{'已写入' if state == 'written' else '已存在且相同'} {a.output}", flush=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
@pytest.mark.parametrize("stage", ("pilot", "full"))
def test_main_stage_paths(stage, tmp_path):
    n, updates, seeds = dict(pilot=(200, 2000, (1703,)), full=(2000, 10000, (1703, 1704, 1705)))[stage]
    names = list(SCENES)
    root = lambda s: V6 if (stage, s) == ("pilot", "double_well") else f"../../features-v7/{s}-{n}"
    feature = lambda argv: {root(s): s for s in names}[opt(argv, "--features-root")]
    by = {s["name"]: [shlex.split(c) for c in s["commands"]] for s in sections(stage, tmp_path)}

    def opt(argv, key):
        return argv[argv.index(key)+1] if key in argv else None

    for argv in itertools.chain(*by.values()):
        if argv[1] == "run.py":
            assert opt(argv, "--trajectories") == str(n) and opt(argv, "--updates") == str(updates)
            system, _, knobs = feature(argv).partition("~")
            assert opt(argv, "--system") == system and opt(argv, "--variant") == (knobs or None)
            assert opt(argv, "--device") == ("cpu" if argv[2] == "spectral-gap" else "cuda:0")
            assert opt(argv, "--output").startswith(f"../../results-addon/{stage}/")
    assert {opt(a, "--features-root") for a in by["C"]} == {root(s) for s in names}
    d = {(opt(a, "--features-root"), int(opt(a, "--seed")), opt(a, "--condition")): opt(a, "--output") for a in by["D"]}
    assert d == {(root(s), seed, c): f"../../results-addon/{stage}/{s}/seed{seed}/{c}"
                 for s in names for seed in seeds for c in ADDON}
    d4 = [a for a in by["D4"] if a[2] == "d4"]
    assert {(opt(a, "--features-root"), int(opt(a, "--seed")), opt(a, "--condition")) for a in d4} == {
        (root(s), seed, c) for s in names for seed in seeds for c in (*V7, *ADDON)}
    for a in d4:
        scene, seed, c = feature(a), opt(a, "--seed"), opt(a, "--condition")
        origin = "results-v7" if c in V7 else "results-addon"
        assert opt(a, "--training-run") == f"../../{origin}/{stage}/{scene}/seed{seed}/{c}"
        assert opt(a, "--output") == f"../../results-addon/{stage}/d4/{scene}/seed{seed}/{c}"
        assert (opt(a, "--ac-checkpoint") == jobs.AC) == (c == "vjepa2_ac") and (opt(a, "--ac-checkpoint") is None) == (c != "vjepa2_ac")
    assert [opt(a, "--output") for a in by["D4"] if a[2] == "spectral-gap"] == [f"../../results-addon/{stage}/spectral-gap/{s}" for s in names]
    text = "\n".join(lines(stage))
    if stage == "pilot":     # double_well 的 C 1、D 5、D4 11、谱间隙 1 条读 v6 目录，preflight 也拿到它
        assert "features-v7/double_well" not in text and text.count(f"--features-root {V6} ") == 18
        assert f"--features ../../features-v7 --v6-features {V6} --stage pilot" in text and "results-v7/pilot" in text
    else:
        assert "--v6-features" not in text and V6 not in text

def accepts(entry, command, capsys):
    """run.py 的子命令选项里有没有 command：有就打印帮助、退出码 0，没有就是“无效的选项”、退出码 2。"""
    with pytest.raises(SystemExit) as error:
        entry.main([command, "--help"])
    capsys.readouterr()
    assert error.value.code in (0, 2)
    return error.value.code == 0


def test_every_command_is_accepted_without_running(tmp_path, monkeypatch, capsys):
    """每条 run.py 命令过 run.py 的参数检查（子进程换成记录器，不执行）；本段的两个工具和 plan_v7.py 过各自的参数解析。
    d4、spectral-gap 是 S18 的子命令：run.py 已经有就要求通过，还没有就要求恰好报“无效的子命令”。"""
    work = tmp_path/"full-model"/"po-jepa-full"
    work.mkdir(parents=True)
    monkeypatch.chdir(work)
    for name in ("vjepa2-vitg-fpc64-256.pth", "vjepa2-ac-vitg.pt"):
        (tmp_path/"checkpoints").mkdir(exist_ok=True)
        (tmp_path/"checkpoints"/name).write_bytes(b"")
    entry = load_run_py()
    s18 = accepts(entry, "d4", capsys)                                 # S18 合并以后 run.py 才有 d4、spectral-gap
    assert accepts(entry, "spectral-gap", capsys) == s18
    seen = []
    monkeypatch.setattr(entry.subprocess, "run", lambda command, **kw: seen.append(command) or Namespace(returncode=0))
    import addon_preflight
    import import_v7_calibration
    import plan_v7
    monkeypatch.setattr(import_v7_calibration, "import_calibration", lambda *a, **k: {})
    monkeypatch.setattr(addon_preflight, "preflight", lambda *a, **k: dict(scenes={}, checked_jobs=0))
    monkeypatch.setattr(addon_preflight, "write_once", lambda *a, **k: "written")
    counted = dict(run=0, tool=0, s18=0)
    v6 = str(tmp_path/"v6-features")                                   # 绝对路径，但在临时目录里（下面会建出来）
    listing = {st: jobs.job_lines(st, v6 if st == "pilot" else "") for st in ("pilot", "full", "variants")}
    for stage in ("pilot", "full", "variants"):
        for line in listing[stage]:
            if line.startswith("#"):
                continue
            argv = shlex.split(line)
            assert argv[0] == "python"
            for key in ("--features-root", "--training-run"):     # run.py 要求特征目录存在；S18 的子命令也许要求训练目录存在
                if key in argv:
                    Path(argv[argv.index(key)+1]).mkdir(parents=True, exist_ok=True)
            if argv[1] == "run.py":
                seen.clear()
                if argv[2] in ("d4", "spectral-gap") and not s18:
                    with pytest.raises(SystemExit) as error:
                        entry.main(argv[2:])
                    assert error.value.code == 2
                    counted["s18"] += 1
                    continue
                assert entry.main(argv[2:]) == 0, line
                child = seen[0][seen[0].index("--")+1:]
                module = child[child.index("-m")+1]
                if argv[2] in ("calibrate-addon", "check-addon"):
                    assert module == "po_jepa.runtime.addon.entry"
                elif argv[2] not in ("d4", "spectral-gap"):            # S18 的两个子命令交给 S18 定的模块
                    assert module == "po_jepa.runtime.cli"
                counted["run"] += 1
            elif argv[1] == "tools/import_v7_calibration.py":
                assert import_v7_calibration.main(argv[2:]) == 0
                counted["tool"] += 1
            elif argv[1] == "tools/addon_preflight.py":
                assert addon_preflight.main(argv[2:]) == 0
                counted["tool"] += 1
            elif argv[1] == "tools/plan_v7.py":
                monkeypatch.setattr(sys, "argv", argv[1:])
                parsed = plan_v7._parse()
                assert parsed.no_baselines and parsed.samples == 256 and parsed.elites == 24 and parsed.job.name in ADDON
                counted["tool"] += 1
            else:
                assert argv[1] in ("tools/encoder_identity.py", "tools/summarize_addon.py")   # S18 的工具
    total_run = sum(1 for st in listing for line in listing[st] if line.startswith("python run.py"))
    assert counted["run"]+counted["s18"] == total_run == 5+1+25+60+5+1+75+170+150
```
