```python
"""08 §4：固定 Fourier 对照的谱模块——二维 DCT-II 基底和解析特征值，替换 PO-JEPA 的自适应图谱。

FourierSpectrum 继承 GraphSpectrum，不加参数、不加 buffer（q、sigma_x、sigma_z 照常构造和保存，
构造时消耗的随机数相同），只重写 spectrum_for：基底和特征值只由网格和 sigma_x 决定，
source 只用来核形状、batch 和设备，数值不参与计算。
"""

import math

from dataclasses import dataclass

import torch
from torch import Tensor

from .dynamics import DynamicsConfig, IntegratorConfig, ModalDynamics
from .geometry import GridSpec, regular_grid
from .model import POJEPA, ProjectionConfig
from .spectral import GraphConfig, GraphSpectrum, SpectralDiagnostics, SpectralResult


@dataclass(frozen=True)
class FourierConfig(GraphConfig):
    """GraphConfig 只多一个字段 basis（只接受 "dct2"）；training._model_config 的 graph 快照因此和 full 不同。"""

    basis: str = "dct2"

    def __post_init__(self) -> None:
        """先做 GraphConfig 的全部检查，再要求 basis == "dct2"。"""
        super().__post_init__()
        if self.basis != "dct2":
            raise ValueError("FourierConfig.basis must be 'dct2'")


def mode_order(count: int) -> tuple:
    """08 §4.3：按 (p²+q², max(p,q), p) 升序的前 count 个 (p, q)，与网格无关。

    p 是 x 方向频率，q 是 y 方向频率。前 count 个模态的 p、q 都小于 count（(0,0)…(0,count−1)
    已有 count 个，p²+q² 都小于 count²），所以只需在 [0,count)² 里排序。
    """
    if type(count) is not int or count <= 0:
        raise ValueError("mode count must be a positive integer")
    candidates = [(p, q) for p in range(count) for q in range(count)]
    return tuple(sorted(candidates, key=lambda mode: (mode[0]**2+mode[1]**2, max(mode), mode[0]))[:count])


def dct_basis(height: int, width: int, modes) -> Tensor:
    """08 §4.3：u_pq(x,y) = c_p^W c_q^H cos(πp(x+.5)/W) cos(πq(y+.5)/H)，token n = y·W + x。

    c_0^N = √(1/N)，c_p^N = √(2/N)（p > 0）。返回 CPU float64 [H·W, len(modes)]，各列在 H·W 个 token 上正交归一。
    """
    if any(type(v) is not int or v <= 0 for v in (height, width)):
        raise ValueError("grid height/width must be positive integers")
    modes = tuple(modes)
    if not modes or any(type(m) is not tuple or len(m) != 2 or any(type(v) is not int or v < 0 for v in m) for m in modes):
        raise ValueError("modes must be a nonempty sequence of (p, q) nonnegative integer pairs")
    if any(p >= width or q >= height for p, q in modes):
        raise ValueError("grid does not resolve the requested Fourier modes")
    y, x = torch.meshgrid(torch.arange(height, dtype=torch.float64, device="cpu"),
                          torch.arange(width, dtype=torch.float64, device="cpu"), indexing="ij")
    x, y = x.reshape(-1), y.reshape(-1)
    columns = []
    for p, q in modes:
        cp = math.sqrt((1. if p == 0 else 2.)/width)
        cq = math.sqrt((1. if q == 0 else 2.)/height)
        columns.append(cp*cq*torch.cos(math.pi*p*(x+.5)/width)*torch.cos(math.pi*q*(y+.5)/height))
    return torch.stack(columns, 1)


def fourier_eigenvalues(modes, sigma_x_squared) -> Tensor:
    """08 §4.4：λ_pq = 1 − exp(−σ_x² π² (p²+q²) / 4)，CPU float64 [len(modes)]；与网格大小无关。"""
    s = torch.tensor([p*p+q*q for p, q in modes], dtype=torch.float64, device="cpu")
    return 1-torch.exp(-sigma_x_squared*math.pi**2*s/4)


class FourierSpectrum(GraphSpectrum):
    """08 §4.2：固定 DCT-II 基底的谱模块；不加参数和 buffer，只重写 spectrum_for。"""

    def __init__(self, config: FourierConfig) -> None:
        """要求 FourierConfig，其余和 GraphSpectrum 完全相同（q 用独立的 CPU generator，不消耗全局随机数）。"""
        if not isinstance(config, FourierConfig):
            raise TypeError("config must be a FourierConfig")
        super().__init__(config)

    def spectrum_for(self, source: Tensor, grid: GridSpec, *, k: int,
                     model_device: torch.device) -> SpectralResult:
        """08 §4.5：返回和 GraphSpectrum.lowest_eigenpairs 同类型、同形状、同精度的 SpectralResult。

        source 只核形状、batch、设备和有限值（和 affinity 的输入检查相同）；grid 必须是规则格点中心；
        前 min(k+1, N) 个模态在这个网格上都要能表示（第 k+1 个用于 cutoff_gap）。
        """
        if not isinstance(source, Tensor) or not isinstance(grid, GridSpec):
            raise TypeError("x must be a tensor and grid must be a GridSpec")
        grid.__post_init__()
        width = self.config.feature_dim
        if source.ndim != 3 or source.shape[0] < 1 or source.shape[1] != grid.height*grid.width or source.shape[2] != width:
            raise ValueError(f"x must have shape [positive B, grid height*width, {width}]")
        if source.dtype != torch.float32 or source.device.type == "meta" or not torch.isfinite(source).all():
            raise ValueError("x must be a finite float32 tensor on a real model device")
        _, sx2, _ = self._working_buffers()
        nodes = grid.height*grid.width
        if type(k) is not int or not 1 <= k <= nodes:
            raise ValueError("k must be an integer in [1,N]")
        if not isinstance(model_device, torch.device):
            raise TypeError("model_device must be a torch.device")
        if model_device.type not in ("cpu", "cuda"):
            raise ValueError("first-round model_device must be CPU or available CUDA")
        if model_device.type == "cuda":
            if not torch.cuda.is_available():
                raise ValueError("requested CUDA is unavailable")
            index = torch.cuda.current_device() if model_device.index is None else model_device.index
            if not 0 <= index < torch.cuda.device_count():
                raise ValueError("requested CUDA index does not exist")
            model_device = torch.device("cuda", index)
        if not torch.allclose(grid.coords, regular_grid(grid.height, grid.width).coords, atol=1e-12, rtol=0.):
            raise ValueError("fixed Fourier basis requires regular cell-center coordinates")
        modes = mode_order(min(k+1, nodes))
        if any(p >= grid.width or q >= grid.height for p, q in modes):
            raise ValueError("grid does not resolve the first k+1 Fourier modes")
        batch = source.shape[0]
        with torch.no_grad():
            values = fourier_eigenvalues(modes, sx2)
            basis_cpu = dct_basis(grid.height, grid.width, modes[:k])[None].repeat(batch, 1, 1)
            eigenvalues_cpu = values[:k][None].repeat(batch, 1)
            residuals = torch.zeros(batch, k, dtype=torch.float64, device="cpu")
            gram_error = basis_cpu.transpose(-1, -2) @ basis_cpu - torch.eye(k, device="cpu", dtype=torch.float64)[None]
            orthogonality = torch.linalg.matrix_norm(gram_error, ord="fro", dim=(-2, -1))
            gaps = eigenvalues_cpu[:, 1:]-eigenvalues_cpu[:, :-1]
            cutoff = (values[k]-values[k-1]).repeat(batch) if k < nodes else None
            pivots = basis_cpu.abs().argmax(dim=1)
            diagnostics = SpectralDiagnostics(residuals, orthogonality, gaps, cutoff, pivots)
            if any(not torch.isfinite(t).all() for t in (basis_cpu, eigenvalues_cpu, orthogonality, gaps)) or (cutoff is not None and not torch.isfinite(cutoff).all()):
                raise ValueError("spectral diagnostics must contain finite values")
            return SpectralResult(eigenvalues_cpu, basis_cpu,
                                  eigenvalues_cpu.to(device=model_device, dtype=torch.float32).detach(),
                                  basis_cpu.to(device=model_device, dtype=torch.float32).detach(), diagnostics)


def build_fixed_fourier(model_config: dict, seed: int, device: torch.device) -> POJEPA:
    """08 §1.2、§4.2：和 checks.build_full_model 相同的顺序和种子，只把 GraphSpectrum 换成 FourierSpectrum。

    model_config 是含 projection/graph/dynamics/integrator 四块的模型配置（仿真传 load_protocol()["model"]，
    DROID 传 09 §2.1 的配置块）；seed 是初始化种子（仿真传 checks.model_seed()）。构造在 fork_rng 里做，不改全局随机状态。
    """
    if type(seed) is not int:
        raise TypeError("seed must be an int")
    m = model_config
    with torch.random.fork_rng(devices=[]):
        torch.random.default_generator.manual_seed(seed)
        model = POJEPA(ProjectionConfig(**m["projection"]), FourierSpectrum(FourierConfig(**m["graph"], basis="dct2")),
            ModalDynamics(DynamicsConfig(**m["dynamics"])), IntegratorConfig(**m["integrator"]))
    return model.to(device)
```

- `choose_modes(po_parameters, action_dim=…, encoder_dim=1408)`：从 K = 1 开始，下一个 K 的差严格更小才往上走（参数量随 K 严格递增，差先减后增），所以相等时停在较小的 K（08 §5.6）。m = 2、PO-JEPA 215,778 → 76；m = 7、236,898 → 72。
- 参数创建顺序 `p_in`、`p_out`（xavier_uniform gain 1、偏置 0，和 POJEPA 相同）、`basis`（16×16 上前 K 个 DCT-II 模态，转 float32）、`raw_gamma`（常数 log(expm1(0.1))）、`omega`（0）、`action_gain`（Normal(0, 0.001)）；只有 `p_in`、`p_out`、`action_gain` 消耗随机数。`named_parameters()` 的顺序是 `basis, raw_gamma, omega, action_gain, p_in.weight, p_in.bias, p_out.weight, p_out.bias`（`nn.Parameter` 直接属性在子模块之前，这是 PyTorch 的规则）。没有 buffer。
- `grid_basis`：16×16 时要求坐标和 `regular_grid(16,16)` 逐位相同，返回 `basis` 本身；其他网格用 `build_spatial_transfer(regular_grid(16,16), …)` 重采样（梯度传回 `basis`），要求 `transfer.target.coords` 和输入网格坐标逐位相同（08 §5.3 第 1 步；§9 第 3、4 条）。
- `analysis`：c_0 = (256/N_h)·W_hᵀX；16×16 上乘的是 1.0，所以和 WᵀX 逐位相同。
- `integrate`：`clip_action_segments(controls, start=0, end=T)`（和 `TokenTransformer` 相同）取 [0, T) 内的有效段；各段长之和和 T 的绝对差超过 1e-12 报错（08 §5.3 第 4 步）；每对通道 (2i, 2i+1) 配成复数（`view_as_complex`），λ = −softplus(raw_gamma) + i·ω 在 float64 上算（保留梯度），e^{λΔ} 和 Δ·φ1(λΔ) 在 complex128 上算、乘之前转 complex64；某一行没有第 j 段时用 `torch.where` 原样保留，所以 T = 0 时 c_T 与 c_0 逐位相同；padding 段的动作值不读（`where` 先置 0）。
- `phi1(w, threshold)`：|w| < threshold 用 1 + w/2 + w²/6，否则 `expm1(w)/w`（§9 第 5 条）；两个分支都先换成安全参数，未选分支的 0/0 不会把 NaN 带进反向。
- `forward`：Ẑ = Z + (P_out(W_h c_T) − P_out(W_h c_0))，`p_out` 调用两次，括号的写法和 POJEPA 的残差读出相同（T = 0 时差是精确的 0，输出逐位等于 Z）。在残差读出下 `p_out.bias` 的梯度恒为 0（两次读出相消，POJEPA 也一样，§9 第 10 条）。
- `build_koopman(config, seed, device)`：`fork_rng(devices=[])` 里 `manual_seed(seed)` 后构造（和 `comparison._make_transformer` 相同），不动全局随机状态。

```python
"""08 §5：Koopman 对照——可学空间基底上的模态系数，逐对阻尼旋转加线性动作项，按段精确积分，残差读出。

模态之间没有相互作用，γ、ω 不随上下文变化，只用 JEPA 损失训练。输入输出合同和 TokenTransformer.forward 相同。
"""

import math

from dataclasses import dataclass

import torch
from torch import Tensor, nn
from torch.nn import functional as F

from .actions import ActionSegments, clip_action_segments, validate_action_segments
from .fourier import dct_basis, mode_order
from .geometry import GridSpec, apply_spatial_transfer, build_spatial_transfer, regular_grid

BASIS_HEIGHT, BASIS_WIDTH = 16, 16
BASIS_TOKENS = BASIS_HEIGHT*BASIS_WIDTH


@dataclass(frozen=True)
class KoopmanConfig:
    """08 §5.7：Koopman 的模型配置；检查点的 model_config 存它的快照。"""

    modes: int
    action_dim: int
    encoder_dim: int = 1408
    dynamic_dim: int = 64
    basis_grid: tuple = (16, 16)
    basis_init: str = "dct2"
    gamma_init: float = 0.1
    action_gain_std: float = 0.001
    phi1_series_threshold: float = 1e-4
    readout: str = "residual"

    def __post_init__(self) -> None:
        """分配参数前拒绝非法维度和偏离 08 §5 的固定选择。"""
        for name in ("modes", "action_dim", "encoder_dim"):
            value = getattr(self, name)
            if type(value) is not int or value <= 0:
                raise ValueError(f"{name} must be a positive integer")
        if type(self.dynamic_dim) is not int or self.dynamic_dim != 64:
            raise ValueError("dynamic_dim must be 64 (32 complex channel pairs)")
        if type(self.basis_grid) is not tuple or self.basis_grid != (BASIS_HEIGHT, BASIS_WIDTH):
            raise ValueError("basis_grid must be the tuple (16, 16)")
        if self.basis_init != "dct2" or self.readout != "residual":
            raise ValueError("Koopman uses the dct2 basis initialization and the residual readout")
        for name in ("gamma_init", "action_gain_std"):
            value = getattr(self, name)
            if type(value) is not float or not math.isfinite(value) or value <= 0:
                raise ValueError(f"{name} must be a finite positive float")
        if type(self.phi1_series_threshold) is not float or self.phi1_series_threshold != 1e-4:
            raise ValueError("phi1_series_threshold must be 1e-4")


def parameter_count(config: KoopmanConfig) -> int:
    """08 §5.2：2·E·D + D + E + K·(256 + D + D·m)；E=1408、D=64 时为 181,696 + K·(320 + 64m)。"""
    if not isinstance(config, KoopmanConfig):
        raise TypeError("config must be a KoopmanConfig")
    e, d, k, m = config.encoder_dim, config.dynamic_dim, config.modes, config.action_dim
    return 2*e*d+d+e+k*(BASIS_TOKENS+d+d*m)


def choose_modes(po_parameters: int, *, action_dim: int, encoder_dim: int = 1408) -> int:
    """08 §5.6：使 |parameter_count(K) − PO-JEPA 参数量| 最小的正整数 K，相等时取较小的 K。"""
    if type(po_parameters) is not int or po_parameters <= 0:
        raise ValueError("po_parameters must be a positive integer")
    def difference(k):
        return abs(parameter_count(KoopmanConfig(modes=k, action_dim=action_dim, encoder_dim=encoder_dim))-po_parameters)
    k = 1
    while difference(k+1) < difference(k):   # 参数量随 K 严格递增，差先减后增
        k += 1
    return k


def phi1(w: Tensor, threshold: float) -> Tensor:
    """φ1(w) = (e^w − 1)/w：|w| < threshold 时用 1 + w/2 + w²/6，否则用 expm1(w)/w。

    两个分支都先换成安全参数（未选分支的 0/0 不会把 NaN 带进反向），再按 |w| 选。
    """
    small = w.abs() < threshold
    ratio_arg = torch.where(small, torch.ones_like(w), w)
    series_arg = torch.where(small, w, torch.zeros_like(w))
    ratio = torch.expm1(ratio_arg)/ratio_arg
    series = 1+series_arg/2+series_arg*series_arg/6
    return torch.where(small, series, ratio)


def _validate_inputs(raw: Tensor, controls: ActionSegments, duration: Tensor, grid: GridSpec,
                     config: KoopmanConfig, device: torch.device) -> None:
    """与 baselines.validate_inputs 相同的输入合同，只把动作维 2 换成 config.action_dim。"""
    if (not isinstance(raw, Tensor) or raw.layout != torch.strided or raw.is_nested or raw.is_meta
        or raw.dtype != torch.float32 or raw.ndim != 3 or min(raw.shape) <= 0
        or raw.shape[-1] != config.encoder_dim or raw.device != device or not torch.isfinite(raw).all()):
        raise ValueError("raw must be finite model-device float32 [B,N,encoder_dim]")
    if not isinstance(grid, GridSpec):
        raise TypeError("grid must be GridSpec")
    grid.__post_init__()
    if raw.shape[1] != grid.height*grid.width:
        raise ValueError("raw N must match actual grid")
    validate_action_segments(controls, query_duration=duration)
    if controls.u_values.shape[0] != raw.shape[0] or controls.u_values.device != device:
        raise ValueError("control B/device must match raw")
    if controls.control_spec.action_dim != config.action_dim:
        raise ValueError("control channels must equal the Koopman action_dim")


class KoopmanPredictor(nn.Module):
    """08 §5：Φ(Z, u, T; h) = Z + P_out(W_h c_T) − P_out(W_h c_0)。"""

    def __init__(self, config: KoopmanConfig) -> None:
        """按 08 §5.2 的表的顺序创建参数；只有 p_in、p_out、action_gain 消耗调用者的 torch 随机数。"""
        super().__init__()
        if not isinstance(config, KoopmanConfig):
            raise TypeError("config must be a KoopmanConfig")
        config.__post_init__()
        self.config = config
        self.encoder_dim, self.dynamic_dim, self.readout = config.encoder_dim, config.dynamic_dim, config.readout
        self.p_in = nn.Linear(config.encoder_dim, config.dynamic_dim, bias=True, device="cpu", dtype=torch.float32)
        self.p_out = nn.Linear(config.dynamic_dim, config.encoder_dim, bias=True, device="cpu", dtype=torch.float32)
        for layer in (self.p_in, self.p_out):
            nn.init.xavier_uniform_(layer.weight, gain=1.)
            nn.init.zeros_(layer.bias)
        pairs = config.dynamic_dim//2
        self.basis = nn.Parameter(dct_basis(BASIS_HEIGHT, BASIS_WIDTH, mode_order(config.modes)).to(dtype=torch.float32))
        self.raw_gamma = nn.Parameter(torch.full((config.modes, pairs), math.log(math.expm1(config.gamma_init)),
                                                 dtype=torch.float32, device="cpu"))
        self.omega = nn.Parameter(torch.zeros(config.modes, pairs, dtype=torch.float32, device="cpu"))
        self.action_gain = nn.Parameter(torch.empty(config.modes, config.dynamic_dim, config.action_dim,
                                                    dtype=torch.float32, device="cpu"))
        nn.init.normal_(self.action_gain, mean=0., std=config.action_gain_std)

    def grid_basis(self, grid: GridSpec) -> Tensor:
        """08 §5.3 第 1 步：W_h [N_h, K]。16×16 上就是 basis；其他网格用和 token 相同的重采样 R（梯度传回 basis）。

        16×16 的坐标必须和 regular_grid(16,16) 逐位相同；其他网格的坐标必须和 R 的目标坐标逐位相同。
        """
        source = regular_grid(BASIS_HEIGHT, BASIS_WIDTH)
        if (grid.height, grid.width) == (BASIS_HEIGHT, BASIS_WIDTH):
            if not torch.equal(grid.coords, source.coords):
                raise ValueError("grid coordinates must equal the 16x16 basis grid or its spatial transfer")
            return self.basis
        transfer = build_spatial_transfer(source, target_height=grid.height, target_width=grid.width)
        if not torch.equal(transfer.target.coords, grid.coords):
            raise ValueError("grid coordinates must equal the 16x16 basis grid or its spatial transfer")
        return apply_spatial_transfer(self.basis[None], transfer)[0]

    def analysis(self, x: Tensor, basis: Tensor) -> Tensor:
        """08 §5.3 第 2 步：c_0 = (256 / N_h) · W_hᵀ X，[B, K, 64]；16×16 上就是 Wᵀ X（乘 1.0 是精确的）。"""
        return torch.matmul(basis.transpose(0, 1), x)*(BASIS_TOKENS/basis.shape[0])

    def integrate(self, initial: Tensor, controls: ActionSegments, duration: Tensor) -> Tensor:
        """08 §5.3 第 3、4 步：每对通道 z = c[2i] + i·c[2i+1]，λ = −softplus(raw_gamma) + i·ω，逐段
        z ← e^{λΔ} z + Δ·φ1(λΔ)·φ，φ = B·u 同样配对。e^{λΔ}、Δ·φ1(λΔ) 用 complex128 算，乘之前转 complex64。

        段取 [0, T) 内的有效段（clip_action_segments，和 TokenTransformer 相同），Δ_j = 段长；
        padding 段不读，行里没有第 j 段时该行原样保留，所以 T = 0 时 c_T 与 c_0 逐位相同。
        """
        clipped = clip_action_segments(controls, start=torch.zeros_like(duration), end=duration)
        valid = clipped.segment_valid
        lengths = torch.where(valid, clipped.ends-clipped.starts, torch.zeros_like(clipped.ends))
        total = lengths.sum(1)
        if any(abs(float(t)-float(d)) > 1e-12 for t, d in zip(total, duration)):   # 08 §5.3 第 4 步：绝对差 1e-12
            raise ValueError("effective segment lengths must sum to the query duration")
        batch, modes, width = initial.shape
        state = torch.view_as_complex(initial.reshape(batch, modes, width//2, 2))
        rate = torch.complex(-F.softplus(self.raw_gamma.to(dtype=torch.float64)), self.omega.to(dtype=torch.float64))
        device = initial.device
        for j in range(valid.shape[1]):
            active = valid[:, j].to(device)
            if not bool(active.any()):
                continue
            delta = lengths[:, j].to(device=device)[:, None, None]
            w = rate[None]*delta
            propagator = torch.exp(w).to(dtype=torch.complex64)
            forcing_integral = (delta*phi1(w, self.config.phi1_series_threshold)).to(dtype=torch.complex64)
            action = torch.where(valid[:, j, None].to(device), clipped.u_values[:, j], torch.zeros_like(clipped.u_values[:, j]))
            forcing = torch.einsum("kcm,bm->bkc", self.action_gain, action)
            phi = torch.view_as_complex(forcing.reshape(batch, modes, width//2, 2))
            state = torch.where(active[:, None, None], propagator*state+forcing_integral*phi, state)
        return torch.view_as_real(state).reshape(batch, modes, width)

    def forward(self, raw: Tensor, controls: ActionSegments, duration: Tensor, grid: GridSpec) -> Tensor:
        """raw Z [B,N_h,E] 模型 float32，duration 为 CPU float64 [B]（控制周期）→ 同形 raw 预测；p_out 调用两次。"""
        _validate_inputs(raw, controls, duration, grid, self.config, self.p_in.weight.device)
        basis = self.grid_basis(grid)
        initial = self.analysis(self.p_in(raw), basis)
        final = self.integrate(initial, controls, duration)
        output = self.p_out(torch.matmul(basis, final))
        # 偏置在两项中精确相消；c_T = c_0 时差为 0，输出逐位等于输入 Z（和 POJEPA 的残差读出相同的写法）。
        output = raw+(output-self.p_out(torch.matmul(basis, initial)))
        if not torch.isfinite(output).all():
            raise ValueError("nonfinite Koopman raw prediction")
        return output


def build_koopman(config: KoopmanConfig, seed: int, device: torch.device) -> KoopmanPredictor:
    """08 §1.2、§5.2：fork_rng 里用 seed 设种子后构造（和 comparison._make_transformer 相同），再转到 device。

    仿真传 KoopmanConfig(modes=choose_modes(PO 参数量, action_dim=2), action_dim=2) 和 checks.model_seed()。
    """
    if type(seed) is not int:
        raise TypeError("seed must be an int")
    with torch.random.fork_rng(devices=[]):
        torch.random.default_generator.manual_seed(seed)
        model = KoopmanPredictor(config)
    return model.to(device)
```

```python
def closed_form(z, rate, segments):
    """z ← e^{λΔ}z + (e^{λΔ} − 1)/λ · φ，逐段（complex128，和被测代码无关）。"""
    for delta, phi in segments:
        z = cmath.exp(rate*delta)*z+(cmath.exp(rate*delta)-1)/rate*phi
    return z

    λ = −log 2 − 0.5i，φ = i·2·u_x。第 1 行 3 段、第 0 行 2 段，padding 是 NaN，不能被读到。"""

    initial = torch.zeros(2, 2, 64)
    for row in (0, 1):

    for row, segments in rows.items():
```

### 4.1 `po_jepa.fourier`

```python
@dataclass(frozen=True)
class FourierConfig(GraphConfig):            # 只多 basis: str = "dct2"
def mode_order(count: int) -> tuple[tuple[int, int], ...]
def dct_basis(height: int, width: int, modes) -> Tensor            # CPU float64 [H·W, len(modes)]
def fourier_eigenvalues(modes, sigma_x_squared) -> Tensor           # CPU float64 [len(modes)]
class FourierSpectrum(GraphSpectrum):
    def spectrum_for(self, source: Tensor, grid: GridSpec, *, k: int, model_device: torch.device) -> SpectralResult
def build_fixed_fourier(model_config: dict, seed: int, device: torch.device) -> POJEPA
```

- `build_fixed_fourier`：`model_config` 是含 `projection`、`graph`、`dynamics`、`integrator` 四块的 dict（仿真传 `load_protocol()["model"]`）；`graph` 块里不能已有 `basis` 键。返回的模型 `type(model.spectrum) is FourierSpectrum`，`state_dict` 的键和值与同一种子的 `build_full_model` 逐项相同。
- `spectrum_for` 对网格的要求：坐标和 `regular_grid(H, W)` 在 1e-12 内相同；前 min(k+1, N) 个模态的 p < W、q < H。仿真的 16×16、8×8（平均池化目标）、12×12（双线性目标）都满足；8×8 上 k = 16 时要前 17 个模态，(4,0) 的 p = 4 < 8，满足。
- 不调用 `affinity`、`normalized_laplacian`、`lowest_eigenpairs`；返回张量都不带梯度。

### 4.2 `po_jepa.koopman`

```python
@dataclass(frozen=True)

def phi1(w: Tensor, threshold: float) -> Tensor
```
